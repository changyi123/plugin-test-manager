```typescript
// POST /webtriggers/api-automation-webhook
export async function automationWebhook(params) {
  const payload = params.payload;
  
  // 1. 提取分支信息
  const branch = extractBranchFromRef(payload.ref);
  const targetBranch = process.env.SYNC_BRANCH || 'master';
  
  // 2. 分支过滤
  if (branch !== targetBranch) {
    return buildResponse({ 
      success: true, 
      message: 'Non-target branch, skipped' 
    });
  }
  
  // 3. 存储到队列
  const queueItem = {
    id: generateId(),
    webhook_uuid: payload.uuid || payload.hook_id,
    repository_id: payload.project.id,
    repository_name: payload.project.name,
    branch_name: branch,
    commit_ids: payload.commits.map(c => c.id),
    payload: payload,
    status: 'pending',
    retry_count: 0,
    created_time: new Date()
  };
  
  await saveToQueue(queueItem);
  
  // 4. 立即返回
  return buildResponse({ 
    success: true, 
    queueId: queueItem.id 
  });
}
```

### 2.3 测试代码解析
```typescript
// 解析Java测试文件
function parseJavaTestFile(content: string, diff: any) {
  const testCases = [];
  
  // 正则匹配@TestId注解和测试方法
  const testPattern = /@TestId\s*\(\s*"([^"]+)"\s*\)\s*\n\s*public\s+void\s+(\w+)\s*\(/g;
  
  let match;
  while ((match = testPattern.exec(content)) !== null) {
    const [fullMatch, testId, methodName] = match;
    const lineNumber = getLineNumber(content, match.index);
    
    // 检查该方法是否在diff范围内
    if (isLineInDiff(lineNumber, diff)) {
      testCases.push({
        automationCaseId: testId,
        name: methodName,
        testMethodName: methodName,
        lineNumber: lineNumber
      });
    }
  }
  
  // 提取类名
  const classMatch = content.match(/public\s+class\s+(\w+)/);
  const className = classMatch ? classMatch[1] : 'Unknown';
  
  return testCases.map(tc => ({
    ...tc,
    testClassName: className
  }));
}

// 判断行是否在diff中
function isLineInDiff(lineNumber: number, diff: any) {
  // diff格式: @@ -old_start,old_length +new_start,new_length @@
  const hunks = diff.diff.match(/@@ .+? \+(\d+),(\d+) @@/g);
  
  for (const hunk of hunks) {
    const [, start, length] = hunk.match(/\+(\d+),(\d+)/);
    const startLine = parseInt(start);
    const endLine = startLine + parseInt(length);
    
    if (lineNumber >= startLine && lineNumber <= endLine) {
      return true;
    }
  }
  
  return false;
}
```

### 2.4 用例同步逻辑
```typescript
// 同步单个测试用例
async function syncTestCase(testCase: any, moduleId: string) {
  // 1. 构建IQL查询
  const iql = `'自动化用例唯一标识' = "${testCase.automationCaseId}"`;
  
  // 2. 查询是否存在
  const existingCases = await requestCoreApi('POST', '/parse/api/search', {
    iql: iql,
    displayContext: 'test_manager'
  });
  
  const existingCase = existingCases?.payload?.items?.[0];
  
  // 3. 准备用例数据
  const caseData = {
    name: testCase.name,
    moduleId: moduleId,
    customFields: {
      automationCaseId: testCase.automationCaseId,
      automationFilePath: testCase.filePath,
      automationRepository: testCase.repositoryName,
      automationClassName: testCase.testClassName,
      automationMethodName: testCase.testMethodName,
      automationLastCommit: testCase.commitId,
      automationLastSyncTime: new Date(),
      automationFramework: testCase.testingFramework
    }
  };
  
  // 4. 创建或更新
  if (existingCase) {
    // 更新用例
    await requestCoreApi('PUT', `/testcases/${existingCase.id}`, caseData);
    return { operation: 'updated', caseId: existingCase.id };
  } else {
    // 创建用例
    const newCase = await requestCoreApi('POST', '/testcases', caseData);
    return { operation: 'created', caseId: newCase.id };
  }
}

// 获取模块ID
async function getModuleIdByPath(modulePath: string) {
  // modulePath格式: "全部用例库/用户认证/登录功能"
  const parts = modulePath.split('/');
  let currentModuleId = null;
  
  for (const part of parts) {
    const iql = currentModuleId 
      ? `name = "${part}" AND parentId = "${currentModuleId}"`
      : `name = "${part}" AND parentId = null`;
    
    const modules = await requestCoreApi('POST', '/parse/api/search', {
      iql: iql,
      displayContext: 'test_manager_modules'
    });
    
    const module = modules?.payload?.items?.[0];
    if (!module) {
      console.log(`Module not found: ${part} in path ${modulePath}`);
      return null;
    }
    
    currentModuleId = module.id;
  }
  
  return currentModuleId;
}
```

```typescript
// src/trigger/modules/job/automationSync.ts
export async function processAutomationQueue() {
  console.log('[AutomationSync] Starting queue processing...');
  
  try {
    // 1. 获取待处理记录
    const pendingRecords = await getPendingWebhooks(5);
    
    if (pendingRecords.length === 0) {
      console.log('[AutomationSync] No pending records');
      return;
    }

      await processSingleWebhook(record);
    }
    
    console.log('[AutomationSync] Queue processing completed');
  } catch (error) {
    console.error('[AutomationSync] Error:', error);
  }
}
```

## 五、错误处理策略

### 5.1 重试机制
- 失败后最多重试3次
- 重试间隔：第1次30秒，第2次1分钟，第3次2分钟
- 超过重试次数标记为failed

### 5.2 错误分类
1. **可重试错误**
   - 网络超时
   - API限流
   - 临时服务不可用

2. **不可重试错误**
   - 配置文件不存在
   - 文件解析失败
   - 权限不足

### 5.3 错误记录
所有错误记录在sync_log表的error_details字段中，包含：
- 错误类型
- 错误消息
- 错误堆栈
- 失败的具体用例
