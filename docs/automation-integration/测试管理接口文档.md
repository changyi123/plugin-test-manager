9. 文件上传接口（分为3步骤）
步骤一
地址：http://{{Host}}/forge/files/${Encoding(fileName)}
示例：http://${host}/forge/files/%E8%BF%99%E6%98%AF%E4%B8%80%E4%B8%AA%E6%96%87%E4%BB%B6.txt
将文件流写入 http body 中即可。
返回200，返回如下

}
Java代码示例，使用 OkHttp 上传文件：
package org.example;

import okhttp3.*;

import java.io.File;
import java.io.IOException;

public class FileUploadDemo {

    public static void main(String[] args) {
        OkHttpClient client = new OkHttpClient();

        File file = new File("test.xml"); // 指定上传文件的路径

        // 创建RequestBody封装文件
        RequestBody requestBody = new MultipartBody.Builder()
                .setType(MultipartBody.FORM)
                .addFormDataPart("file", file.getName(), RequestBody.create(MediaType.parse("application/octet-stream"), file))
                .build();

        // 创建POST请求
        Request request = new Request.Builder()
                .url("http://team-host/forge/files/test.xml")
                .post(requestBody)
                .addHeader("X-Parse-Application-Id", "osc")
                .addHeader("X-Proxima-API-Token", "a:ug9yu2qzh84rl102az")
                .build();

        // 发送请求并处理响应
        try (Response response = client.newCall(request).execute()) {
            if (response.isSuccessful()) {
                System.out.println("File uploaded successfully");
                System.out.println("response: " + response.body().string());
            } else {
                System.out.println("File upload failed. HTTP Code: " + response.code());
            }
        } catch (IOException e) {
            e.printStackTrace();
        }
    }
}
